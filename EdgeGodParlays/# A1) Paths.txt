# A3) Tighten ACLs (only your user & Administrators)
$acl = Get-Acl $ROOT
$ruleUser = New-Object System.Security.AccessControl.FileSystemAccessRule("$env:USERNAME","FullControl","ContainerInherit,ObjectInherit","None","Allow")
$ruleAdmins = New-Object System.Security.AccessControl.FileSystemAccessRule("Administrators","FullControl","ContainerInherit,ObjectInherit","None","Allow")
$acl.SetAccessRuleProtection($true,$false)
$acl.ResetAccessRule($ruleUser)
$acl.AddAccessRule($ruleAdmins)
Set-Acl -Path $ROOT -AclObject $acl
